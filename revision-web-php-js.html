1. Validation de formulaire (pattern récurrent)

JavaScript — Gabarit Validation
function check() {
    var erreurs = "";

    // 1. Champ texte obligatoire
    var champ = document.getElementById("id_champ").value;
    if (champ == "" || champ == null) {
        erreurs += "Le champ X est obligatoire.\n";
    }

    // 2. Vérification format (ex: CIN = 8 chiffres)
    if (champ.length != 8 || isNaN(champ)) {
        erreurs += "Le CIN doit contenir 8 chiffres.\n";
    }

    // 3. Radio button obligatoire
    var radios = document.getElementsByName("pack");
    var packChoisi = false;
    var valeurPack = "";
    for (var i = 0; i < radios.length; i++) {
        if (radios[i].checked) {
            packChoisi = true;
            valeurPack = radios[i].value;
        }
    }
    if (!packChoisi) {
        erreurs += "Veuillez choisir un pack.\n";
    }

    // 4. Select obligatoire
    var niveau = document.getElementById("niveau").value;
    if (niveau == "*" || niveau == "") {
        erreurs += "Veuillez sélectionner un niveau.\n";
    }

    // 5. Règle métier : si niveau = TIC → pas de spécialité
    var spec = document.getElementById("spec").value;
    if (niveau == "TIC" && spec != "") {
        erreurs += "Un étudiant TIC ne peut pas choisir une spécialité.\n";
    }

    // 6. Affichage ou soumission
    if (erreurs != "") {
        alert(erreurs);
        return false;
    }
    return true;
}
✅ Emplacement dans le formulaire :
<form onsubmit="return check()"> ou sur le bouton submit : <button type="submit" onclick="return check()">

3. Calcul de frais (CalculFrais — examens 2024-2025)

JavaScript — CalculFrais
function CalculFrais() {
    // Récupérer le pack sélectionné
    var radios = document.getElementsByName("pack");
    var fraisPack = 0;
    var isSingle = false;
    for (var i = 0; i < radios.length; i++) {
        if (radios[i].checked) {
            fraisPack = parseInt(radios[i].value);
            if (radios[i].id == "packSingle") isSingle = true;
        }
    }

    // Récupérer frais optionnels
    var checkboxes = document.getElementsByName("FO");
    var fraisOptAvant = 0;
    for (var j = 0; j < checkboxes.length; j++) {
        if (checkboxes[j].checked) {
            fraisOptAvant += parseInt(checkboxes[j].value);
        }
    }

    // Réduction 20% si pack Single
    var fraisOptApres = isSingle ? fraisOptAvant * 0.8 : fraisOptAvant;
    var fraisTotal = fraisPack + fraisOptApres;

    var msg = "Les Frais sont :\n";
    msg += "- Frais total : " + fraisTotal + "€\n";
    msg += "- Frais du pack : " + fraisPack + "€\n";
    msg += "- Frais optionnels (avant réduction) : " + fraisOptAvant + "€\n";
    msg += "- Frais optionnels (après réduction) : " + fraisOptApres + "€";
    alert(msg);
}

4. Affichage récapitulatif (show / message)

JavaScript — Récapitulatif
function show() {
    var nom    = document.getElementById("nom").value;
    var prenom = document.getElementById("pr").value;
    var msg = "Le mannequin \"" + nom + "\" \"" + prenom + "\" est affecté à :\n";

    // Robes sélectionnées (select multiple)
    var selectRobe = document.getElementById("robe");
    for (var i = 0; i < selectRobe.options.length; i++) {
        if (selectRobe.options[i].selected && selectRobe.options[i].value != "f") {
            msg += "  - \"" + selectRobe.options[i].value + "\"\n";
        }
    }

    // Participations (checkboxes)
    msg += "Pendant :\n";
    var participations = document.getElementsByName("p");
    for (var j = 0; j < participations.length; j++) {
        if (participations[j].checked) {
            msg += "  - \"" + participations[j].value + "\"\n";
        }
    }
    alert(msg);
}

⚠️ Erreurs Fréquentes en JS

🚫 Erreurs à éviter :
• Oublier return false dans la fonction de validation → le formulaire se soumet quand même
• Utiliser getElementById pour un name (il faut getElementsByName pour radio/checkbox)
• Oublier de boucler sur les radios/checkboxes pour trouver le coché
• Confondre .value (texte/select) et .checked (radio/checkbox)
• Pour un select multiple : utiliser .options[i].selected et non .checked
📌 Règle métier — Pattern récurrent :
Toujours du type : "Si l'utilisateur a choisi X comme [sexe/niveau/filière], il [ne doit pas / doit] sélectionner Y"
→ Vérifier avec if (valX == "X" && valY != "") ou if (valX == "X" && !yEstCoché)

// Exemple — mostWatched (examen 2021-2022) : films réservés > 100 fois
function mostWatched($dateD, $dateF) {
    global $conn;
    $sql = "SELECT Nom_Film, COUNT(*) AS nbr FROM RESERVATION
            WHERE DateRes BETWEEN '$dateD' AND '$dateF'
            GROUP BY Nom_Film
            HAVING COUNT(*) > 100";
    $result = mysqli_query($conn, $sql);
    while ($row = mysqli_fetch_assoc($result)) {
        echo $row['Nom_Film'] . "<br>";
    }
}

// Exemple — RevenuParZone (examen 2022-2023)

class NomClasse {

class Etudiant {
    private $nom;
    private $cin;
    private $email;

    public function __construct($nom, $cin, $email) {

               . $this->email . "')";

}

🔧 Héritage

PHP OOP — Héritage (extends)
class Personne {
    protected $nom;
    protected $email;

    public function __construct($nom, $email) {
        $this->nom   = $nom;
        $this->email = $email;
    }

    public function afficher() {
        echo "Personne : " . $this->nom . "<br>";
    }
}

// Classe enfant
class Formateur extends Personne {
    private $specialite;

    public function __construct($nom, $email, $specialite) {
        parent::__construct($nom, $email); // appel constructeur parent
        $this->specialite = $specialite;
    }

    public function afficher() { // surcharge de la méthode
        parent::afficher();  // appel méthode parent
        echo "Spécialité : " . $this->specialite . "<br>";
    }
}

$f = new Formateur("Sami", "sami@tek.tn", "Web");
$f->afficher();
✅ Règle private vs protected :
• private → accessible seulement dans la classe elle-même
• protected → accessible dans la classe ET les classes filles
• public → accessible partout
⚠️ Si une classe fille doit utiliser un attribut du parent dans sa méthode → utiliser protected !

// Exemple complet — Liste réservations
function listeReservationsPDO($num_client, $dateD, $dateF) {
    global $pdo;
    $stmt = $pdo->prepare(
        "SELECT Nom_Film, NumSalle FROM RESERVATION

         AND DateRes BETWEEN :d1 AND :d2"

    $stmt->execute([':nc' => $num_client, ':d1' => $dateD, ':d2' => $dateF]);

    echo "Mes réservations entre $dateD et $dateF :<ol>";
    while ($row = $stmt->fetch(PDO::FETCH_ASSOC)) {
        echo "<li>" . $row['Nom_Film'] . ", salle " . $row['NumSalle'] . "</li>";
    }
    echo "</ol>";
}

mysqli — style procédural

  • mysqli_connect(host, user, pass, db)
  • mysqli_query($conn, $sql)
  • mysqli_fetch_assoc($result)
  • Pas de requêtes préparées obligatoires
  • Plus simple, style des anciens examens

PDO — style objet

  • new PDO("mysql:host=...;dbname=...")
  • $pdo->prepare($sql)
  • $stmt->execute([':param' => $val])
  • $stmt->fetch(PDO::FETCH_ASSOC)
  • Plus sécurisé, style recommandé

📌 Différence Session vs Cookie :
• Session : stockée côté serveur, dure jusqu'à la fermeture du navigateur, identifiée par un ID
• Cookie : stocké côté client (navigateur), peut durer longtemps, visible dans le navigateur

PARTIE 6 📋 Cheat Sheet — Tout en Un

📌 Must Know — Concepts Indispensables

  • 1
    session_start() doit être appelé avant tout echo/HTML dans chaque fichier PHP qui utilise les sessions.
  • 2
    $_SESSION['numClient'] est toujours utilisé dans l'INSERT — ne jamais l'oublier dans la réponse d'examen.
  • 3
    date("Y-m-d H:i:s") pour la date actuelle en PHP. En SQL c'est NOW().
  • 4
    L'état par défaut est toujours "non confirmée" dans les examens — à insérer littéralement dans l'INSERT.
  • 5
    return false dans la fonction JS de validation empêche la soumission du formulaire. Sans ça, la validation ne sert à rien.
  • 6
    getElementsByName() retourne un tableau — il faut boucler avec for pour lire les radios/checkboxes.
  • 7
    En OOP, les attributs private ne sont pas accessibles dans les classes filles — utiliser protected si nécessaire.
  • 8
    PDO utilise prepare + execute avec des paramètres nommés (:param) pour éviter les injections SQL.

Question 1 — ValidationReservation()

JavaScript — CORRECTION Q1
function ValidationReservation() {
    var erreurs = "";

    // Vérifier Email
    var email = document.getElementById("email").value;
    if (email == "") {
        erreurs += "L'email est obligatoire.\n";
    }

    // Vérifier sport sélectionné
    var sport = document.getElementById("sport").value;
    if (sport == "*" || sport == "") {
        erreurs += "Veuillez choisir un sport.\n";
    }

    // Vérifier pack choisi
    var radios = document.getElementsByName("pack");
    var packChoisi = false;
    for (var i = 0; i < radios.length; i++) {
        if (radios[i].checked) packChoisi = true;
    }
    if (!packChoisi) {
        erreurs += "Veuillez choisir un pack.\n";
    }

    // Règle métier : Tennis → pas de Coach
    var coach = document.getElementById("coach");
    if (sport == "Tennis" && coach.checked) {
        erreurs += "Le service Coach n'est pas disponible pour le Tennis.\n";
    }

    if (erreurs != "") {
        alert(erreurs);
        return false;
    }
    return true;
}

Question 2 — CalculFrais()

JavaScript — CORRECTION Q2
function CalculFrais() {
    // Lire le pack
    var radios = document.getElementsByName("pack");
    var fraisPack = 0;
    var est2h = false;
    for (var i = 0; i < radios.length; i++) {
        if (radios[i].checked) {
            fraisPack = parseInt(radios[i].value);
            if (radios[i].id == "pack2h") est2h = true;
        }
    }

    // Lire les services optionnels
    var services = document.getElementsByName("svc");
    var fraisOpt = 0;
    for (var j = 0; j < services.length; j++) {
        if (services[j].checked) {
            fraisOpt += parseInt(services[j].value);
        }
    }

    // Réduction 10% si pack 2h
    var fraisOptFinal = est2h ? fraisOpt * 0.9 : fraisOpt;
    var total = fraisPack + fraisOptFinal;

    var msg = "Les Frais sont :\n"
            + "- Frais total : " + total + " DT\n"
            + "- Frais du pack : " + fraisPack + " DT\n"
            + "- Frais optionnels : " + fraisOptFinal + " DT";
    alert(msg);
}

function mesReservations($num_client, $dateD, $dateF) {
    global $conn;
    $sql = "SELECT NomSport, NumTerrain FROM RESERVATION
            WHERE NumClient = '$num_client'
            AND EtatRes = 'confirmée'
            AND DateRes BETWEEN '$dateD' AND '$dateF'";
    $result = mysqli_query($conn, $sql);

    echo "Mes réservations entre $dateD et $dateF sont :<ul>";
    while ($row = mysqli_fetch_assoc($result)) {
        echo "<li>" . $row['NomSport'] . " : " . $row['NumTerrain'] . "</li>";
    }
    echo "</ul>";
}

function sportPopulaire($dateD, $dateF) {
    global $conn;
    $sql = "SELECT NomSport, COUNT(*) AS total FROM RESERVATION
            WHERE DateRes BETWEEN '$dateD' AND '$dateF'
            GROUP BY NomSport
            HAVING COUNT(*) > 50";
    $result = mysqli_query($conn, $sql);

    echo "Sports populaires :<ul>";
    while ($row = mysqli_fetch_assoc($result)) {
        echo "<li>" . $row['NomSport'] . "</li>";
    }
    echo "</ul>";
}
